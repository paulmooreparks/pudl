function parsePlacement(value) {
  var m = /^([a-z]+):([0-9.]+),([0-9.]+),([0-9.]+),([0-9.]+)$/.exec(value || '');
  if (!m) return null;
  var p = { mode: m[1], x: +m[2], y: +m[3], w: +m[4], h: +m[5] };
  return validPlacement(p) ? p : null;
}
